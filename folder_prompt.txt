def classify(rel_path: str, item_type: str):
    rel = rel_path.replace('\\', '/')

    if rel in FORMAL_ROOTS:
        role, scope, status = FORMAL_ROOTS[rel]
        return role, scope, status, 'yes' if status.startswith('current') else 'reference', 'no', 'listed formal/reference root'

    if rel in SPECIAL_ROOTS:
        role, scope, status = SPECIAL_ROOTS[rel]
        return role, scope, status, 'no' if role in {'LEGACY', 'CACHE'} else 'reference', 'no', 'special known root'

    top = rel.split('/')[0]

    if '/_archived' in rel or rel.startswith('runs/_archived') or rel.startswith('outputs/_backup'):
        return 'ARCHIVE', 'archived or backup material', 'historical', 'no', 'no', 'archive/backup path'

    if '__pycache__' in rel or rel.endswith('.pyc'):
        return 'CACHE', 'python cache', 'ignore', 'no', 'no', 'cache path'

    if rel.startswith('outputs/ib3') or rel.startswith('scripts/ib3') or rel.startswith('weather'):
        return 'FUTURE_LAYER', 'IB3 activity/behavior/weather layer', 'future_or_legacy', 'no', 'no', 'outside current IA1-IB2D checkpoint'

    if rel.startswith('outputs/ib4') or rel.startswith('scripts/ib4') or rel.startswith('outputs/model_'):
        return 'EXPERIMENTAL', 'IB4/personal/model artifacts', 'historical_or_future', 'no', 'no', 'outside current IA1-IB2D checkpoint'

    if rel.startswith('scripts/ia_nlsc') or rel.startswith('scripts/ia2'):
        return 'REFERENCE', 'older NLSC/IA2 scripts', 'legacy_reference', 'reference', 'no', 'not current IB1G/IB1E runner'

    if rel.startswith('outputs/') and '_v1_3b_contract_qa' not in rel and 'convergence_audit' not in rel:
        return 'LEGACY_OR_EXPERIMENTAL_OUTPUT', 'older/non-current output root', 'review_before_use', 'no', 'no', 'not listed as current formal root'

    if top in {'activity_input', 'configs', 'nlsc_raw', 'osm_raw_output', 'outputs', 'runs', 'scripts'}:

inventory_rows = []
role_rows = []

for p in ROOT.rglob('*'):
    rel = p.relative_to(ROOT)

    if any(part in EXCLUDE_DIRS for part in rel.parts):
        continue

    depth = len(rel.parts)
    if depth > MAX_DEPTH:
        continue

    rel_str = str(rel).replace('\\', '/')

    try:
        stat = p.stat()
        size = stat.st_size if p.is_file() else ''
        mtime = datetime.fromtimestamp(stat.st_mtime).strftime('%Y-%m-%d %H:%M:%S')
    except Exception:
        size = ''
        mtime = ''

    inventory_rows.append({
        'relative_path': rel_str,
        'name': p.name,
        'type': 'dir' if p.is_dir() else 'file',
        'depth': depth,
        'size_bytes': size,
        'modified_time': mtime,
        'suffix': p.suffix.lower() if p.is_file() else '',
    })

    if depth <= 2 or rel_str in FORMAL_ROOTS or rel_str in SPECIAL_ROOTS:
        role, scope, status, used, blocker, reason = classify(rel_str, 'dir' if p.is_dir() else 'file')
        role_rows.append({
            'relative_path': rel_str,
            'name': p.name,
            'type': 'dir' if p.is_dir() else 'file',
            'depth': depth,
            'role': role,
            'stage_or_scope': scope,
            'formal_status': status,
            'used_by_current_pipeline': used,
            'current_pipeline_blocker': blocker,
            'reason': reason,
